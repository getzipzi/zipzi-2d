cmake_minimum_required(VERSION 3.20)
project(kasane_core LANGUAGES CXX)
add_library(kasane_core STATIC src/document.cpp src/geometry.cpp src/legacy_deformers.cpp src/keyforms.cpp src/evaluation.cpp)
target_include_directories(kasane_core PUBLIC include PRIVATE ../purism-core/include)
target_compile_features(kasane_core PUBLIC cxx_std_20)
set_target_properties(kasane_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(kasane_moc3 STATIC src/moc3.cpp)
target_link_libraries(kasane_moc3 PUBLIC kasane_core)
set_target_properties(kasane_moc3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(kasane_core_tests tests/core_tests.cpp)
  target_link_libraries(kasane_core_tests PRIVATE kasane_core)
  add_test(NAME kasane_core_tests COMMAND kasane_core_tests)
  add_executable(kasane_keyform_tests tests/keyform_tests.cpp)
  target_link_libraries(kasane_keyform_tests PRIVATE kasane_core)
  add_test(NAME kasane_keyform_tests COMMAND kasane_keyform_tests)
  add_executable(kasane_deformer_tests tests/deformer_tests.cpp)
  target_link_libraries(kasane_deformer_tests PRIVATE kasane_core)
  add_test(NAME kasane_deformer_tests COMMAND kasane_deformer_tests)
endif()

# Optional compatibility harness. The core/codec themselves have no Core ABI
# dependency. Each provider runs in a separate process to avoid symbol mixing.
option(KASANE_MOC3_CONFORMANCE "Build MOC3 Core compatibility tests" OFF)
set(KASANE_CUBISM_ROOT "" CACHE PATH "Local Cubism Native SDK root")
if(BUILD_TESTING AND KASANE_MOC3_CONFORMANCE)
  enable_language(C)
  if(NOT TARGET PurismCore)
    add_subdirectory(../purism-core ${CMAKE_CURRENT_BINARY_DIR}/purism)
  endif()
  add_executable(kasane_moc3_purism_tests tests/moc3_tests.cpp)
  target_link_libraries(kasane_moc3_purism_tests PRIVATE kasane_moc3 PurismCore)
  add_test(NAME kasane_moc3_purism_tests COMMAND kasane_moc3_purism_tests)
  if(KASANE_CUBISM_ROOT)
    if(APPLE)
      set(_cubism_platform macos)
    elseif(UNIX)
      set(_cubism_platform linux)
    else()
      message(FATAL_ERROR "Set up an official Core library for this platform before conformance")
    endif()
    set(_cubism_library "${KASANE_CUBISM_ROOT}/Core/lib/${_cubism_platform}/${CMAKE_SYSTEM_PROCESSOR}/libLive2DCubismCore.a")
    if(NOT EXISTS "${_cubism_library}")
      message(FATAL_ERROR "Missing official Core: ${_cubism_library}")
    endif()
    add_executable(kasane_moc3_official_tests tests/moc3_tests.cpp)
    target_compile_definitions(kasane_moc3_official_tests PRIVATE KASANE_OFFICIAL_CORE)
    target_include_directories(kasane_moc3_official_tests PRIVATE "${KASANE_CUBISM_ROOT}/Core/include")
    target_link_libraries(kasane_moc3_official_tests PRIVATE kasane_moc3 "${_cubism_library}")
    add_test(NAME kasane_moc3_official_tests COMMAND kasane_moc3_official_tests)
  endif()
endif()
